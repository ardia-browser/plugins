        this.#updating = false;
      }
    }

      try {
        answer = JSON.parse(second);
      } catch (e) {
        answer = { ok: false, error: "the daemon's answer couldn't be read" };
      }
      settle(first, answer);
    } else if (kind === "timer") fire(first);
    else if (kind === "run") run(first);
  };
})
