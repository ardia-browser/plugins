      if (typeof given === "function") throw cloneFailure(`${given.name || "function"}`);
      if (given === null || typeof given !== "object") return given;

      let out;

        case "[object Boolean]":

        case "[object Number]":

        case "[object String]":

        case "[object BigInt]":

          seen.set(given, out);
          return out;
        case "[object ArrayBuffer]":
          out = ArrayBuffer.prototype.slice.call(given, 0);
          seen.set(given, out);
          return out;
        case "[object Map]":
          out = new Map();
          seen.set(given, out);
          for (const [key, item] of Map.prototype.entries.call(given)) out.set(copy(key), copy(item));
          return out;
        case "[object Set]":
          out = new Set();
          seen.set(given, out);
          for (const item of Set.prototype.values.call(given)) out.add(copy(item));
          return out;
        case "[object Array]":
          out = new Array(given.length);
          seen.set(given, out);
          return copyProps(given, out);
        case "[object Error]": {

          seen.set(given, out);

        }

            out = tag === "[object DataView]"
              ? new DataView(buffer, given.byteOffset, given.byteLength)
              : new given.constructor(buffer, given.byteOffset, given.length);
            seen.set(given, out);
            return out;
          }
          if (tag !== "[object Object]") throw cloneFailure(tag.slice(8, -1));
          out = {};
          seen.set(given, out);
          return copyProps(given, out);
      }
    };
    return copy(value);
